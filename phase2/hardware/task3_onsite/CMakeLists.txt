cmake_minimum_required(VERSION 3.16)
project(airsign_task3_onsite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Franka 0.20 REQUIRED)
find_package(Threads REQUIRED)
add_executable(native_arm_translation tested/native_arm_translation.cpp)
target_link_libraries(native_arm_translation PRIVATE Franka::Franka Threads::Threads)
add_executable(probe_franka_state ../../scripts/probe_franka_state.cpp)
target_link_libraries(probe_franka_state PRIVATE Franka::Franka)
option(AIRSIGN_BUILD_TMR "Build against the site's TMR-extended libfranka" OFF)
if(AIRSIGN_BUILD_TMR)
  find_package(nlohmann_json REQUIRED)
  add_executable(native_scene_turn tested/native_scene_turn.cpp)
  target_link_libraries(native_scene_turn PRIVATE Franka::Franka Threads::Threads nlohmann_json::nlohmann_json)
  add_executable(read_base_feedback tested/read_base_feedback.cpp)
  target_link_libraries(read_base_feedback PRIVATE Franka::Franka)
endif()
