cmake_minimum_required(VERSION 3.16)
project(airsign_gamepad LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(gamepad_mock native_worker.cpp)
target_link_libraries(gamepad_mock PRIVATE Threads::Threads)
option(AIRSIGN_BUILD_LIVE_GAMEPAD "Build experimental live worker against site libfranka" OFF)
if(AIRSIGN_BUILD_LIVE_GAMEPAD)
  find_package(Franka 0.20 REQUIRED)
  add_executable(gamepad_franka native_worker.cpp)
  target_compile_definitions(gamepad_franka PRIVATE AIRSIGN_LIVE=1)
  target_link_libraries(gamepad_franka PRIVATE Franka::Franka Threads::Threads)
endif()
